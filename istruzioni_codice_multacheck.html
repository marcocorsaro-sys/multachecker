-- =============================================
-- MULTACHECK — SCHEMA COMPLETO v1.0
-- Eseguire nell'editor SQL di Supabase Dashboard
-- =============================================

-- Estensioni necessarie
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- PROFILI UTENTE (estende auth.users)
CREATE TABLE profiles (
  id            UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  email         TEXT NOT NULL,
  phone         TEXT,
  pec_address   TEXT,           -- PEC provisionata via Namirial
  has_firma     BOOLEAN DEFAULT false,
  plan          TEXT DEFAULT 'free' CHECK (plan IN ('free','advise','managed')),
  stripe_customer_id TEXT,
  created_at    TIMESTAMPTZ DEFAULT now(),
  updated_at    TIMESTAMPTZ DEFAULT now()
);

-- VERBALI
CREATE TABLE verbali (
  id                  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id             UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  file_path           TEXT,
  raw_ocr             JSONB,
  parsed              JSONB NOT NULL DEFAULT '{}',
  contestability      JSONB,
  semaforo            TEXT CHECK (semaforo IN ('verde','giallo','rosso')),
  score               INT CHECK (score BETWEEN 0 AND 100),
  status              TEXT DEFAULT 'parsed' CHECK (
                        status IN ('parsed','paid','ricorso_generato','inviato','esito')
                      ),
  -- Campi estratti per query veloci
  tipo_infrazione     TEXT,     -- velocita|sosta|ztl|semaforo|telefono|altro
  importo_base        NUMERIC(8,2),
  importo_ridotto     NUMERIC(8,2),
  importo_doppio      NUMERIC(8,2),
  punti_decurtati     INT,
  data_infrazione     DATE,
  data_notifica       DATE,
  ente_accertatore    TEXT,
  provincia           CHAR(2),
  -- Scadenze calcolate
  scadenza_sconto     DATE,     -- data_notifica + 5gg (contestazione immediata) o 60gg
  scadenza_prefetto   DATE,     -- data_notifica + 60gg
  scadenza_gdp        DATE,     -- data_notifica + 30gg
  -- Metadata
  parsing_confidence  NUMERIC(3,2),
  created_at          TIMESTAMPTZ DEFAULT now()
);

-- PRATICHE DI RICORSO
CREATE TABLE pratiche (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  verbale_id      UUID NOT NULL REFERENCES verbali(id) ON DELETE CASCADE,
  user_id         UUID NOT NULL REFERENCES profiles(id),
  tipo            TEXT NOT NULL CHECK (tipo IN ('prefetto','gdp','autotutela')),
  tier            TEXT NOT NULL CHECK (tier IN ('advise','managed')),
  stripe_pi       TEXT,           -- Stripe PaymentIntent ID
  importo_pagato  INT,            -- in centesimi
  ricorso_draft   TEXT,
  ricorso_pdf     TEXT,           -- path su Storage
  pec_message_id  TEXT,
  pec_sent_at     TIMESTAMPTZ,
  pec_ricevuta    TEXT,
  esito           TEXT CHECK (esito IN ('accolto','rigettato','silenzio_assenso','pendente')),
  avvocato_id     UUID,
  note_legali     TEXT,
  created_at      TIMESTAMPTZ DEFAULT now(),
  updated_at      TIMESTAMPTZ DEFAULT now()
);

-- AUTOVELOX MIT (sync settimanale)
CREATE TABLE autovelox_omologati (
  id          SERIAL PRIMARY KEY,
  codice_mit  TEXT UNIQUE NOT NULL,
  modello     TEXT,
  produttore  TEXT,
  omologato   BOOLEAN DEFAULT false,
  data_dm     DATE,
  comune      TEXT,
  provincia   CHAR(2),
  note        TEXT,
  updated_at  TIMESTAMPTZ DEFAULT now()
);
CREATE INDEX idx_autovelox_modello ON autovelox_omologati USING gin(to_tsvector('italian', modello));

-- PREFETTURE E PEC UFFICIALI
CREATE TABLE prefetture (
  provincia     CHAR(2) PRIMARY KEY,
  nome          TEXT NOT NULL,
  pec_address   TEXT NOT NULL,
  accetta_pec   BOOLEAN DEFAULT true,
  note          TEXT
);

-- CONSENSI GDPR
CREATE TABLE consents (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id     UUID REFERENCES profiles(id) ON DELETE CASCADE,
  tipo        TEXT NOT NULL,   -- 'privacy_policy' | 'terms' | 'marketing'
  version     TEXT NOT NULL,
  accepted_at TIMESTAMPTZ DEFAULT now(),
  ip_hash     TEXT
);

-- =============================================
-- ROW LEVEL SECURITY
-- =============================================
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE verbali ENABLE ROW LEVEL SECURITY;
ALTER TABLE pratiche ENABLE ROW LEVEL SECURITY;

CREATE POLICY "profiles_owner" ON profiles FOR ALL USING (auth.uid() = id);
CREATE POLICY "verbali_owner" ON verbali FOR ALL USING (auth.uid() = user_id);
CREATE POLICY "pratiche_owner" ON pratiche FOR ALL USING (auth.uid() = user_id);

-- Lettura pubblica autovelox e prefetture (nessun dato personale)
ALTER TABLE autovelox_omologati ENABLE ROW LEVEL SECURITY;
CREATE POLICY "autovelox_public_read" ON autovelox_omologati FOR SELECT USING (true);
ALTER TABLE prefetture ENABLE ROW LEVEL SECURITY;
CREATE POLICY "prefetture_public_read" ON prefetture FOR SELECT USING (true);

-- =============================================
-- TRIGGER: aggiorna updated_at automaticamente
-- =============================================
CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN NEW.updated_at = now(); RETURN NEW; END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER profiles_updated_at BEFORE UPDATE ON profiles
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();
CREATE TRIGGER pratiche_updated_at BEFORE UPDATE ON pratiche
  FOR EACH ROW EXECUTE FUNCTION update_updated_at();

Flusso:
1. Ricevi il file dal FormData
2. Valida: max 10MB, solo image/* o application/pdf
3. Carica il file su Supabase Storage nel bucket "verbali" 
   con path: {userId}/{verbaleId}.{ext}
   (crea URL firmato con scadenza 1 ora per l'elaborazione)
4. Chiama Google Cloud Vision API per OCR:
   - Se PDF: usa document text detection
   - Se immagine: usa text detection
   - Endpoint: POST https://vision.googleapis.com/v1/images:annotate
   - Auth: service account key da env GOOGLE_CLOUD_CREDENTIALS
5. Passa il testo OCR a Claude per parsing strutturato:

   usa generateObject da 'ai' con schema Zod:
   {
     ente_accertatore: z.string(),
     numero_verbale: z.string().optional(),
     targa: z.string().optional(),
     intestatario: z.string().optional(),
     data_infrazione: z.string().optional(), // YYYY-MM-DD
     ora_infrazione: z.string().optional(),
     luogo_infrazione: z.string().optional(),
     articolo_violato: z.string().optional(),
     tipo_infrazione: z.enum(['velocita','sosta','ztl','semaforo','telefono','altro']),
     importo_ridotto: z.number().optional(),
     importo_base: z.number().optional(),
     data_notifica: z.string().optional(), // YYYY-MM-DD
     modalita_notifica: z.string().optional(),
     dispositivo_tipo: z.string().optional(),
     velocita_rilevata: z.number().optional(),
     velocita_limite: z.number().optional(),
     punti_decurtati: z.number().optional(),
     parsing_confidence: z.number().min(0).max(1)
   }

   System prompt Claude:
   "Sei un esperto di diritto stradale italiano. Estrai i campi strutturati 
   dal testo OCR di un verbale di contravvenzione italiano. 
   Per data_infrazione e data_notifica usa formato YYYY-MM-DD.
   Per parsing_confidence dai un valore da 0 a 1 che indica quanto sei 
   sicuro dell'estrazione (0.9+ = molto sicuro, sotto 0.7 = dati incerti)."

6. Calcola le scadenze:
   - scadenza_sconto = data_notifica + 5 giorni (se contravvenzione contestata immediatamente)
     OPPURE data_notifica + 60 giorni × 30% sconto (dipende dall'ente — usa 60gg come default)
   - scadenza_prefetto = data_notifica + 60 giorni
   - scadenza_gdp = data_notifica + 30 giorni
   Usa date-fns per i calcoli.

7. Salva tutto in tabella verbali su Supabase
8. Ritorna { verbaleId, semaforo: null, parsed } al client
   (il semaforo viene calcolato nel passo successivo)

Gestione errori:
- Vision API fallisce → ritorna errore 422 con messaggio "Immagine non leggibile"
- Claude ritorna confidence < 0.6 → salva comunque ma setta un flag needs_manual_review: true
- Timeout dopo 30 secondi → errore 504

PROMPT — incolla in Claude CodeCLAUDE CODE
Crea il motore di contestabilità in packages/core/src/contestability.ts

Il motore è una funzione TypeScript PURA (no side effects, no API calls, testabile).
Prende i dati del verbale e ritorna il risultato di contestabilità.

INTERFACCE da creare:

interface Vizio {
  tipo: string
  titolo: string           // "Notifica tardiva"
  descrizione: string      // spiegazione human-readable
  normativa: string        // "Art. 201 D.Lgs. 285/1992"
  peso: number             // contributo allo score (0-30)
  automatico: boolean      // rilevato automaticamente vs richiede verifica manuale
  probabilita: 'alta' | 'media' | 'bassa'
}

interface ContestabilityInput {
  data_infrazione: string
  data_notifica: string
  tipo_infrazione: string
  importo_base: number
  dispositivo_tipo?: string
  articolo_violato?: string
  numero_verbale?: string
  ente_accertatore?: string
  parsing_confidence: number
  autovelox_omologato?: boolean  // null se non trovato nel DB
}

interface ContestabilityResult {
  semaforo: 'verde' | 'giallo' | 'rosso'
  score: number              // 0-100
  vizi: Vizio[]
  raccomandazione: string    // testo human-readable
  rischio_raddoppio: boolean // sempre true se si va al Prefetto
  conviene_contestare: boolean
  risparmio_potenziale: number  // importo_base se vince
  costo_rischio: number         // importo_base × 2 se perde al Prefetto
}

REGOLE DEL MOTORE (implementa in ordine di peso):

1. NOTIFICA TARDIVA (peso 30 — massimo)
   Se differenza tra data_infrazione e data_notifica > 90 giorni → vizio AUTOMATICO
   Normativa: Art. 201 c. 1 D.Lgs. 285/1992
   Probabilita: alta

2. AUTOVELOX NON OMOLOGATO (peso 28)
   Se tipo_infrazione === 'velocita' E autovelox_omologato === false → vizio AUTOMATICO
   Normativa: Cass. n.10505/2024; Cass. n.7374/2026
   Probabilita: alta
   Se autovelox_omologato === null → aggiungi vizio con automatico: false 
   ("Non siamo riusciti a verificare il dispositivo — approfondire")

3. MANCANZA NUMERO VERBALE (peso 18)
   Se numero_verbale è null/empty → vizio AUTOMATICO
   Normativa: Art. 200 c. 1 lett. a) D.Lgs. 285/1992

4. MANCANZA NORMA VIOLATA (peso 18)
   Se articolo_violato è null/empty → vizio AUTOMATICO

5. MANCANZA ENTE ACCERTATORE (peso 12)
   Se ente_accertatore è null/empty → vizio AUTOMATICO

6. PARSING INCERTO (peso -5, penalità)
   Se parsing_confidence < 0.7 → riduci score, aggiungi nota 
   "Alcuni dati non sono stati letti chiaramente — verifica manuale consigliata"

CALCOLO SEMAFORO:
- score >= 40 → verde (contesta)
- score 15-39 → giallo (valuta attentamente)
- score < 15 → rosso (paga — non conviene)

RACCOMANDAZIONE:
- verde: "Il verbale presenta vizi significativi. Ti consigliamo di contestarlo."
- giallo: "Il verbale presenta alcune irregolarità ma il successo non è garantito. 
           Considera che un ricorso al Prefetto, se perso, comporta il pagamento 
           del doppio dell'importo (€{importo_doppio})."
- rosso: "Non abbiamo trovato vizi rilevanti. Ti consigliamo di pagare entro 
          il {data_scadenza_sconto} per ottenere lo sconto del 30% 
          (risparmi €{risparmio_sconto})."
